def strings(count):
    out = ([LITERAL_0]*int(count))
    return out
def at(s,i):
    if vkf_string_eof(s,i): return LITERAL_1
    return vkf_string_peek_scalar(s,i)
def white(c):
    return c==LITERAL_2  or  c==LITERAL_3  or  c==LITERAL_4  or  c==LITERAL_5
def digit(c):
    return c>=LITERAL_6  and  c<=LITERAL_7
def skip(s,i):
    p = i
    while white(at(s,p)): p = p+1
    return p
def hex(c):
    if digit(c): return n.code(c)-48
    if c>=LITERAL_8  and  c<=LITERAL_9: return n.code(c)-87
    if c>=LITERAL_10  and  c<=LITERAL_11: return n.code(c)-55
    return -1
def quad(s,p):
    value = 0
    i = 0
    while i<4:
        d = hex(at(s,p+i))
        if d<0: return -1
        value = value*16+d
        i = i+1
    return value
def string(s,start):
    p = start+1
    out = LITERAL_12
    while not vkf_string_eof(s,p):
        c = at(s,p)
        if c==LITERAL_13: return Record(value=out,stop=p+1,valid=True)
        if c<LITERAL_14: return Record(value=out,stop=p,valid=False)
        if c!=LITERAL_15:
            out = out+c
            p = int(vkf_utf8_advance(s,p))
            continue
        p = p+1
        escape = at(s,p)
        p = p+1
        if (escape==LITERAL_16  or  escape==LITERAL_17  or  escape==LITERAL_18):
            out = out+escape
            continue
        if escape==LITERAL_19:
            out = out+str(chr(8))
            continue
        if escape==LITERAL_20:
            out = out+str(chr(12))
            continue
        if escape==LITERAL_21:
            out = out+LITERAL_22
            continue
        if escape==LITERAL_23:
            out = out+LITERAL_24
            continue
        if escape==LITERAL_25:
            out = out+LITERAL_26
            continue
        if escape!=LITERAL_27: return Record(value=out,stop=p,valid=False)
        scalar = quad(s,p)
        if (scalar<0  or  (scalar>=56320  and  scalar<=57343)): return Record(value=out,stop=p,valid=False)
        p = p+4
        if scalar>=55296  and  scalar<=56319:
            if (at(s,p)!=LITERAL_28  or  at(s,p+1)!=LITERAL_29): return Record(value=out,stop=p,valid=False)
            low = quad(s,p+2)
            if (low<56320  or  low>57343): return Record(value=out,stop=p,valid=False)
            scalar = 65536+(scalar-55296)*1024+low-56320
            p = p+6
        out = out+str(chr(scalar))
    return Record(value=out,stop=p,valid=False)
def number_stop(s,start):
    p = start
    if at(s,p)==LITERAL_30: p = p+1
    if not digit(at(s,p)): return -1
    if at(s,p)==LITERAL_31: p = p+1
    if at(s,start)==LITERAL_32: start = start+1
    if at(s,start)!=LITERAL_33:
        while digit(at(s,p)): p = p+1
    if at(s,p)==LITERAL_34:
        p = p+1
        if not digit(at(s,p)): return -1
        while digit(at(s,p)): p = p+1
    if (at(s,p)==LITERAL_35  or  at(s,p)==LITERAL_36):
        p = p+1
        if (at(s,p)==LITERAL_37  or  at(s,p)==LITERAL_38): p = p+1
        if not digit(at(s,p)): return -1
        while digit(at(s,p)): p = p+1
    return p
def literal(s,p,word):
    i = 0
    while not vkf_string_eof(word,i):
        if at(s,p+i)!=at(word,i): return False
        i = i+1
    return True
def value(q,depth):
    if not q.valid: return q
    if depth>256:
        q.valid = False
        return q
    q.position = skip(q.source,q.position)
    c = at(q.source,q.position)
    root = q.count
    if root>=len(q.kinds):
        q.valid = False
        return q
    q.count = root+1
    q.ends[root] = q.count
    if c==LITERAL_39:
        parsed = string(q.source,q.position)
        q.kinds[root] = 3
        q.texts[root] = parsed.value
        q.position = parsed.stop
        q.valid = parsed.valid
        return q
    if (c==LITERAL_40  or  digit(c)):
        stop = number_stop(q.source,q.position)
        if stop<0:
            q.valid = False
            return q
        q.kinds[root] = 4
        token = LITERAL_41
        while q.position<stop:
            token = token+at(q.source,q.position)
            q.position = q.position+1
        q.texts[root] = token
        return q
    if (c==LITERAL_42  or  c==LITERAL_43):
        object = c==LITERAL_44
        close = LITERAL_45
        q.kinds[root] = 2
        if object:
            close = LITERAL_46
            q.kinds[root] = 1
        q.position = skip(q.source,q.position+1)
        if at(q.source,q.position)==close:
            q.position = q.position+1
            return q
        while q.valid:
            if object:
                if at(q.source,q.position)!=LITERAL_47:
                    q.valid = False
                    return q
                q = value(q,depth+1)
                q.position = skip(q.source,q.position)
                if (not q.valid  or  at(q.source,q.position)!=LITERAL_48):
                    q.valid = False
                    return q
                q.position = q.position+1
            q = value(q,depth+1)
            if not q.valid: return q
            q.position = skip(q.source,q.position)
            c = at(q.source,q.position)
            if c==close:
                q.position = q.position+1
                q.ends[root] = q.count
                return q
            if c!=LITERAL_49:
                q.valid = False
                return q
            q.position = skip(q.source,q.position+1)
        return q
    if literal(q.source,q.position,LITERAL_50):
        q.kinds[root] = 5
        q.position = q.position+4
        return q
    if literal(q.source,q.position,LITERAL_51):
        q.kinds[root] = 6
        q.position = q.position+5
        return q
    if literal(q.source,q.position,LITERAL_52):
        q.kinds[root] = 7
        q.position = q.position+4
        return q
    q.valid = False
    return q
def parse(source):
    capacity = n.size(source)+1
    q = Record(source=source,position=0,valid=True,kinds=([0]*int(capacity)),ends=([0]*int(capacity)),texts=strings(capacity),count=0)
    q = value(q,0)
    q.position = skip(source,q.position)
    q.valid = q.valid  and  vkf_string_eof(source,q.position)
    return q
def metadata_matches(original,a,root):
    if root<0: return original==LITERAL_53
    if original==LITERAL_54: return False
    if (root>=a.count  or  a.count>len(a.kinds)  or  a.count>len(a.ends)  or  a.count>len(a.texts)  or  a.count>len(a.numbers)): return False
    stop = a.ends[root]
    if (stop<=root  or  stop>a.count): return False
    q = parse(original)
    if (not q.valid  or  q.count!=stop-root): return False
    i = 0
    while i<q.count:
        at = root+i
        kind = q.kinds[i]
        if (kind!=a.kinds[at]  or  q.ends[i]!=a.ends[at]-root): return False
        if kind==3  and  q.texts[i]!=a.texts[at]: return False
        if kind==4:
            if (not n.equal(n.from_token(q.texts[i]),n.from_number(a.numbers[at]))): return False
        i = i+1
    return True
def json_matches(left,right):
    a = parse(left)
    b = parse(right)
    if (not a.valid  or  not b.valid  or  a.count!=b.count): return False
    i = 0
    while i<a.count:
        kind = a.kinds[i]
        if (kind!=b.kinds[i]  or  a.ends[i]!=b.ends[i]): return False
        if kind==3  and  a.texts[i]!=b.texts[i]: return False
        if kind==4:
            if (not n.equal(n.from_token(a.texts[i]),n.from_token(b.texts[i]))): return False
        i = i+1
    return True
