# Check both boundaries before any UTF-8 slicing, including nonzero starts.
scalar_span(source:str,start:int,stop:int)->bit:
    size:text.byte_length(source)
    (start<0 \/ stop<start \/ stop>size)? @:false
    cursor:0
    found:start=0
    cursor<stop?>
        .cursor:text.next(source,cursor)
        cursor=start? .found:true
    @:found /\ cursor=stop

# Restricted literal text only: no TeX semantics, whitespace collapse, active
# characters, dash/quote substitutions or Unicode normalization is inferred.
literal(source:str)->bit:
    source=""? @:false
    cursor:0
    size:text.byte_length(source)
    cursor<size?>
        c:text.at(source,cursor)
        (text.special(c) \/ c="~" \/ c="-" \/ c="`" \/ c="'" \/ c<"!")? @:false
        .cursor:text.next(source,cursor)
    @:true

unique(values:[str])->bit:
